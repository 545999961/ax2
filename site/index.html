arex-terminal
# install and prepare
python3 -m venv .venv
source .venv/bin/activate
pip install -e '.[research]'
python3 -m arex_v2 download --all

# run five tasks
python3 -m arex_v2 research BrowseComp \
  --n 5 --save-path runs/smoke
